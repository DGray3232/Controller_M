# CMakeLists_user.txt — пользовательские исходники и библиотеки,
# не перезаписывается CubeMX

target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE
    ARM_MATH_CM4
    __ARM_FEATURE_DSP=1
)

target_sources(${CMAKE_PROJECT_NAME} PRIVATE
    Core/Src/globals.c
    Core/Src/pid.c
    Core/Src/AHRSAlgorithms.c
    Core/Src/blackbox.c
    Core/Src/optical_flow_compensation.c
    Core/Src/vibration_analysis.c
    Core/Src/calculate_notch_coeffs.c
    Core/Src/median_moving_average_filter.c
    Core/Src/mtf02.c
    Core/Src/remote_control_mavlink.c
)

target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE
    Drivers/CMSIS/DSP/Include
)

target_link_directories(${CMAKE_PROJECT_NAME} PRIVATE
    Drivers/CMSIS/Lib/GCC
)

target_link_libraries(${CMAKE_PROJECT_NAME}
    arm_cortexM4lf_math
)
